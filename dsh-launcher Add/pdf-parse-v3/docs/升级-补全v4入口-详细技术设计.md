```python
# assets/v4/triage.py
import sys, json
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parents[1]))   # assets/ 根（复用 pdf_text_probe）
import pdf_text_probe as ptp            # 复用 _printable_count/_cjk_count/verdict（0.50/0.30）

class TriageError(Exception):
    def __init__(self, kind, detail=""):
        self.kind = kind                # encrypted | locked | zero_page
        super().__init__(f"TriageError({kind}): {detail}")

def _skew_deg(page) -> float:
    """文本行 bbox 主方向角。>1.5° 记 deskew（初值，CALIB MS-4）。返回度数。"""
    import math
    d = page.get_text("dict")
    pts = []
    for block in d.get("blocks", []):
        if block.get("type") != 0: continue
        for line in block.get("lines", []):
            spans = line.get("spans", [])
            if not spans: continue
            b = line.get("bbox", [0,0,0,0])
            if (b[2]-b[0]) >= 1: pts.append((b[0], b[1], b[2], b[1]))  # 行首尾近似水平线
    if len(pts) < 3: return 0.0
    # 主方向 = 多数行夹角（简化：取行尾-行首向量与水平夹角的中位数）
    import statistics
    angs = []
    for x1,y1,x2,_ in pts:
        if x2-x1 < 1: continue
        angs.append(math.degrees(math.atan2(abs(y2_ := 0.0) or 0.0, x2-x1)))  # 占位，Step7 用行内 span 首尾
    return 0.0   # Step 7 编码：用 line bbox 首行/末 span y 差求角，此处给接口

def triage(pdf_path: str, cfg: dict) -> dict:
    """PDF → triage dict（work/triage.json 内容）。异常：TriageError。"""
    import pypdf
    try:
        reader = pypdf.PdfReader(pdf_path)
    except Exception as e:
        raise TriageError("open_failed", str(e))
    encrypted, locked = False, False
    try:
        if getattr(reader, "is_encrypted", False):
            encrypted = True
            try: reader.decrypt("")
            except Exception: pass
    except Exception: pass
    try: n = len(reader.pages)
    except Exception: n = 0; locked = True
    if locked or (encrypted and n == 0):
        raise TriageError("locked", "encrypted/locked 无密码不可读")
    if n == 0:
        raise TriageError("zero_page", "零页 PDF")
    result = {"file": str(Path(pdf_path).resolve()), "page_count": n,
              "encrypted": encrypted, "locked": locked, "degraded": {"zero_page": False, "blank_pages": []},
              "doc_verdict": None, "pages": []}
    # 逐页探测（--all 语义：路由需要每页 verdict）
    import fitz  # PyMuPDF
    doc = fitz.open(pdf_path)
    text_pages, nontext_pages = [], []
    for i, page in enumerate(doc, 1):
        t = page.get_text() or ""
        pr, cjk = ptp._printable_count(t), ptp._cjk_count(t)
        raw = len(t)
        ratio = round(pr/raw, 3) if raw else 0.0
        st = {"raw_len": raw, "printable": pr, "ratio": ratio}
        v = ptp.verdict(st) if raw else "image"      # raw==0 → image（verdict 同）
        st.pop("raw_len", None)
        rec = {"page": i, "verdict": v, "ratio": ratio, "printable": pr, "cjk": cjk,
               "layout_hint": "single", "skew_deg": _skew_deg(page), "dpi_ok": True,
               "intra_regions": []}
        if v == "text":
            text_pages.append(i)
            try: rec["intra_regions"] = detect_intra(page, doc, i, cfg)   # FR-A4（复用 §1.3）
            except Exception: rec["intra_regions"] = []
        else:
            nontext_pages.append(i)
            if raw == 0: result["degraded"]["blank_pages"].append(i)
        result["pages"].append(rec)
    result["doc_verdict"] = "text" if not nontext_pages else ("scanned" if not text_pages else "hybrid")
    return result

def detect_intra(page, doc, page_no, cfg):
    """FR-A4 页内区域探测：调 textlayer.detect_intra_regions（pt 系行列表）。
       Step 7 编码：构造 pt 系 lines（从 page.get_text dict 取 bbox），转调 textlayer。"""
    from textlayer import detect_intra_regions as _dir

    return []
```

## 2. M2 render.py 详细设计

### 2.1 registry 条目字段（work/render/registry.jsonl，每页一条）

```jsonc
{"page": 87, "pdf_rect_pt": [595.0, 842.0], "target_width": 1860,   // target_width=指定值(m-05)
 "dpi": 225, "pixel": [1860, 2714], "scale": [3.1261, 3.2233],
 "render_sha1": "…", "downsampled_from": null}
```
登记纪律：`dpi = round(pixel[0]/pdf_rect_pt[0]×72)`（派生）；`scale = pixel/pdf_rect_pt`（逐轴）。

### 2.2 局部代码

```python
# assets/v4/render.py
import hashlib, json
from pathlib import Path

class RenderError(Exception):
    def __init__(self, kind, detail=""):
        self.kind = kind                # oversize | crop_degenerate
        super().__init__(f"RenderError({kind}): {detail}")

def _render_png(page, dpi):
    import fitz
    mat = fitz.Matrix(dpi/72, dpi/72)
    pix = page.get_pixmap(matrix=mat)
    return pix

def _png_bytes_le10mb(png_path):
    return Path(png_path).stat().st_size <= 10*1024*1024

def render_page(pdf, page_no, out_dir, cfg, target_width=None):
    """单页渲染。target_width 指定优先；缺省由 cfg.render.dpi_default 反推（Step4 §3.2 口径）。
       返回 registry 条目 dict。超 10MB 按 resolution_steps 降档重渲。"""
    page = pdf[page_no - 1]
    rect = page.rect
    pdf_w, pdf_h = rect.width, rect.height
    steps = [cfg["render"].get("dpi_default", 220)] + cfg["render"].get("resolution_steps", [])[1:]
    if target_width is None:
        tw = round(pdf_w * steps[0] / 72.0)
    else:
        tw = target_width
    out = Path(out_dir); out.mkdir(parents=True, exist_ok=True)
    last_err = None
    for dpi in steps:                      # [220,180,150]（去重后）
        tw = round(pdf_w * dpi / 72.0)     # dpi 驱动 → target_width 派生为登记指定值（Step4 §3.2）
        pix = _render_png(page, dpi)
        png = out / f"page_{page_no:04d}.png"
        pix.save(str(png))
        if _png_bytes_le10mb(png):
            px_w, px_h = pix.width, pix.height
            entry = {"page": page_no, "pdf_rect_pt": [round(pdf_w,2), round(pdf_h,2)],
                     "target_width": tw, "dpi": dpi,
                     "pixel": [px_w, px_h],
                     "scale": [round(px_w/pdf_w, 4), round(px_h/pdf_h, 4)],
                     "render_sha1": hashlib.sha1(Path(png).read_bytes()).hexdigest(),
                     "downsampled_from": None}
            return entry
        # 超 10MB：记下刚渲染档位，降档重渲（覆盖 png）
    raise RenderError("oversize", f"page {page_no} 全档位 >10MB")

def render_pages(pdf, page_nos, out_dir, cfg):
    regs = []
    for pno in page_nos:
        try: regs.append(render_page(pdf, pno, out_dir, cfg))
        except RenderError as e:
            if e.kind == "oversize":
                raise                # 整页视觉放弃（冻结 §10：本地管线继续）由编排层处理
            raise
    return regs

def append_registry(reg_path, entries):
    reg_path = Path(reg_path); reg_path.parent.mkdir(parents=True, exist_ok=True)
    with reg_path.open("a", encoding="utf-8") as f:
        for e in entries: f.write(json.dumps(e, ensure_ascii=False) + "\n")

def px_to_pt(box, reg): return [box[0]/reg["scale"][0], box[1]/reg["scale"][1],
                                box[2]/reg["scale"][0], box[3]/reg["scale"][1]]
def pt_to_px(box, reg): return [round(box[0]*reg["scale"][0]), round(box[1]*reg["scale"][1]),
                                round(box[2]*reg["scale"][0]), round(box[3]*reg["scale"][1])]

def crop_png(png_path, box_px, out_path, pad=8):
    from PIL import Image
    im = Image.open(png_path)
    x1,y1,x2,y2 = box_px
    x1 = max(0, x1-pad); y1 = max(0, y1-pad)
    x2 = min(im.width, x2+pad); y2 = min(im.height, y2+pad)
    if x2 <= x1 or y2 <= y1: raise RenderError("crop_degenerate", f"{box_px} on {png_path}")
    im.crop((x1,y1,x2,y2)).save(out_path)
    return {"bbox_px": [x1,y1,x2,y2], "pad": pad}
```

### 2.3 测试点

- 换算自检：`pt_to_px(px_to_pt(b))` 与 b 逐轴差 ≤1px（≥10 随机 bbox，冻结 §3.3）。
- 降采样：合成超大页（宽 5000pt+）→ 断言触发 >10MB → 降档；仍超（构造巨图）→ RenderError(oversize)。

```python
def maybe_offer_degrade(reason, cfg, out, tty=True) -> int:
    """整管线结构性不可跑 → 弹选/中止。仅结构性触发（M1/M2 缺失/依赖白名单不满足）；
       FR-D7（无 key/--no-egress/预算/断网）不经过此函数（页/表级自动兜底）。
       返回：0 用户选 A（重新运行 v4）; 1 用户选 B（显式降级许可：上层凭退出码1+留痕走 v3 流程）;
             2 选 C 取消; 3 非 TTY 中止（不产出、不静默降级）。"""
    offered = ["A: 解决依赖后重跑 v4", "B: 显式降级 v3.0（告知产物差异）", "C: 取消"]
    rec = {"ts": time.time(), "trigger": "structural", "reason": reason,
           "offered": offered, "chosen": None, "context": "tty" if tty else "non-tty",
           "note": "整管线结构性不可跑，禁止无感降级 v3.0"}
    if not tty:
        rec["context"] = "non-tty"; rec["note"] += "；非交互：中止，不产出，不静默降级"
        _write_choice(out, rec)
        print("结构性不可跑（非交互环境）：中止，未产出 v3.0。请人工处理依赖后重跑。", file=sys.stderr)
        return 3
    print("v4 管线结构性不可跑。请选择：")
    for i, opt in enumerate(offered, 1): print(f"  {i}. {opt}")
    while True:
        try: ans = input("输入 1/2/3：").strip()
        except EOFError: rec["chosen"]=None; _write_choice(out, rec); return 3
        if ans == "1": rec["chosen"]="A"; _write_choice(out, rec); return 0
        if ans == "2":
            rec["chosen"]="B"; _write_choice(out, rec)
            print("已选择显式降级 v3.0。差异告知：v3.0 JSON 无网格表格重建（无 rowspan/colspan 网格）、"
                  "无图片索引块、无 lines[] 坐标层；与原 v3 管线产物一致。"
                  "v3 执行体不内联于 orchestrator（v3 为独立工具链，见 SKILL.md §2/§3）；"
                  "上层调用方收到退出码 1 + work/degrade_choice.json(chosen=B) 后，"
                  "方可按 SKILL.md v3 流程（pdf-inspect/pdf-extract/rapid_ocr → pdf2v3 → finalize_v3）产出 v3.0 JSON。",
                  file=sys.stderr)
            return 1   # 显式降级许可码：上层仅凭此码+留痕才可执行 v3 流程
        if ans == "3": rec["chosen"]="C"; _write_choice(out, rec); return 2
```

```python
def build_argparse():
    ap = argparse.ArgumentParser(description="pdf-parse-v3 v4 管线编排")

    # 兼容保留（MS-1，不进 SKILL.md 示例）：
    ap.add_argument("--page-image", action="append", default=[])
    ap.add_argument("--page", type=int, default=None)
    ap.add_argument("--render-reg-json", default=None)
    ap.add_argument("--doc-sha1", default=None)
    return ap
```
